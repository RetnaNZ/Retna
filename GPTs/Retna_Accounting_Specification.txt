4.1 Readable Description — RAW_ALL!M2
=IF(A2:A1000="","",TRIM(F2:F1000&" | "&G2:G1000&" | "&H2:H1000&" | "&I2:I1000))

4.2 Suggested Category (Vendor Rules map) — RAW_ALL!N2

(Keep this mapping in RAW_ALL, not in Expenses.)

=ARRAYFORMULA(IF(A2:A1000="","",
MAP(M2:M1000,LAMBDA(desc,
IF(desc="","",
IFNA(
  INDEX(
    FILTER(
      'Vendor Rules'!$B$2:$B$200,
      'Vendor Rules'!$A$2:$A$200<>"",
      REGEXMATCH(UPPER(desc),UPPER('Vendor Rules'!$A$2:$A$200))
    ),
  1),
"")
))))
)

4.3 Counterparty (human readable) — RAW_ALL!R2

(Uses best available field and avoids masked card tokens.)

=ARRAYFORMULA(
IF(A2:A1000="","",
MAP(F2:F1000,G2:G1000,H2:H1000,I2:I1000,
LAMBDA(details,particulars,code,ref,
  LET(
    cand, IF(details<>"",details, IF(particulars<>"",particulars, IF(code<>"",code, ref))),
    bad, REGEXMATCH(UPPER(cand),"\d{4}-\*{4}-\*{4}-\d{4}") + REGEXMATCH(cand,"\*{2,}"),
    IF(bad,
      IF(code<>"",code, IF(particulars<>"",particulars, IF(ref<>"",ref,""))),
      cand
    )
  )
))
))

4.4 Counterparty Clean (grouping-friendly normalisation) — RAW_ALL!S2
=ARRAYFORMULA(
IF(R2:R1000="","",
TRIM(
  REGEXREPLACE(
    REGEXREPLACE(
      UPPER(R2:R1000),
      "(?i)\b(LTD|LIMITED|PTY|PTE|INC|INC\.|LLC|CO|COMPANY|NZ|NEW ZEALAND)\b",
      ""
    ),
    "[^A-Z0-9 ]",
    ""
  )
))
)

4.5 Counterparty Source — RAW_ALL!U2
=IF(A2:A1000="","",IF(F2:F1000<>"","Details",""))

4.6 Bank Category — RAW_ALL!V2
=ARRAYFORMULA(
IF(A2:A1000="","",
IF(
  REGEXMATCH(LOWER(M2:M1000),"(interest|fee|debit interest|overdraft|unarranged|a/c|account)"),
  "Bank fees / Interest",
  ""
)))

4.7 Bank Direction — RAW_ALL!W2
=IF(A2:A1000="","",IF(K2:K1000>0,"In","Out"))

4.8 Transfer Pair Key — RAW_ALL!Y2
=IF(A2:A1000="","",TEXT(A2:A1000,"yyyy-mm-dd")&"|"&TEXT(ABS(K2:K1000),"0.00"))

4.9 Role — RAW_ALL!T2

(Use the “locked” Role logic as currently in the file.)

=ARRAYFORMULA(
IF(A2:A1000="","",
IFS(
  (
    ((V2:V1000<>"") + REGEXMATCH(LOWER(M2:M1000),"(interest|fee|debit interest|overdraft|unarranged|a/c|account)"))
    * NOT(REGEXMATCH(UPPER(F2:F1000&" "&G2:G1000&" "&M2:M1000),"(DIRECTOR FEE|DIRECTOR|PAY ?SELF|PAYSELF|DRAWING|NATHANIEL)"))
  ),
  "Bank Txn",

  REGEXMATCH(F2:F1000,"06-0901-0812102-\d{2}") + REGEXMATCH(J2:J1000,"06-0901-0812102-\d{2}") + REGEXMATCH(I2:I1000,"06-0901-0812102-\d{2}"),
  "Transfer",

  REGEXMATCH(UPPER(D2:D1000&" "&E2:E1000&" "&F2:F1000&" "&G2:G1000&" "&H2:H1000&" "&I2:I1000&" "&M2:M1000),"(RWT|WHT|WITHHOLD|WITHHOLDING TAX|RESIDENT WITHHOLDING)"),
  "Tax Credit",

  REGEXMATCH(UPPER(D2:D1000&" "&E2:E1000&" "&F2:F1000&" "&G2:G1000&" "&H2:H1000&" "&I2:I1000&" "&M2:M1000),"(ATM|EFT|EFTPOS|EFT-?POS|EFTPOS CASH|POS CASH|CASH ?WITHDRAW(AL)?|CASH ?OUT|CASHOUT|WITHDRAW|LOTUS)"),
  "Cash Movement",

  REGEXMATCH(UPPER(F2:F1000&" "&G2:G1000&" "&M2:M1000),"(PAY ?SELF|PAYSELF|DIRECTOR|DRAWING|NATHANIEL)") + (J2:J1000="12-3038-0373535-50"),
  "Drawing",

  TRUE,
  IF(K2:K1000>0,"Income","Expense")

))
